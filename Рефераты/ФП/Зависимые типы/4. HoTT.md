
```scala
// 1. Структура 1-типа (группоида)
trait Groupoid[A]:
  type Id[x <: A, y <: A]
  def refl[x <: A]: Id[x, x]
  def sym[x <: A, y <: A](p: Id[x, y]): Id[y, x]
  def trans[x <: A, y <: A, z <: A](p: Id[x, y], q: Id[y, z]): Id[x, z]

// 2. Тип окружности S¹
enum S1:
  case Base

object S1:
  // 3. Абстрактный тип путей (петель) на S¹ без привязки к Int!
  // Путь p: Loop — это полиморфная функция, вычисляющаяся в любой целевой группоид.
  trait Loop:
    def apply[A](using G: Groupoid[A])(b: A, l: G.Id[b.type, b.type]): G.Id[b.type, b.type]

  // S¹ само является группоидом (свободным группоидом с одной петлёй)
  given Groupoid[S1] with
    type Id[x <: S1, y <: S1] = Loop

    def refl[x <: S1]: Loop = new Loop:
      def apply[A](using G: Groupoid[A])(b: A, l: G.Id[b.type, b.type]): G.Id[b.type, b.type] =
        G.refl[b.type]

    def sym[x <: S1, y <: S1](p: Loop): Loop = new Loop:
      def apply[A](using G: Groupoid[A])(b: A, l: G.Id[b.type, b.type]): G.Id[b.type, b.type] =
        G.sym(p(b, l))

    def trans[x <: S1, y <: S1, z <: S1](p: Loop, q: Loop): Loop = new Loop:
      def apply[A](using G: Groupoid[A])(b: A, l: G.Id[b.type, b.type]): G.Id[b.type, b.type] =
        G.trans(p(b, l), q(b, l))

  // Канонический конструктор пути: loop : Base = Base
  val loop: Loop = new Loop:
    def apply[A](using G: Groupoid[A])(b: A, l: G.Id[b.type, b.type]): G.Id[b.type, b.type] =
      l

  // 4. ПОЛНОЦЕННЫЙ УНИВЕРСАЛЬНЫЙ РЕКУРСОР (rec_S1)
  // Принимает целевой группоид A, точку b: A и петлю l: b =_A b.
  // Возвращает пару: функцию на точках (S1 => A) и действие на путях (ap_f).
  case class RecResult[A](using G: Groupoid[A])(
    val onPoint: S1 => A,
    val onLoop: Loop => G.Id[?, ?]
  )

  def rec[A](using G: Groupoid[A])(b: A, l: G.Id[b.type, b.type]): (S1 => A, Loop => G.Id[b.type, b.type]) =
    val f: S1 => A = { case S1.Base => b }
    val apF: Loop => G.Id[b.type, b.type] = p => p(b, l)
    (f, apF)
```

// Эквивалентность типов (путь во Вселенной Type согласно унивалентности)
case class Equiv[X, Y](to: X => Y, from: Y => X)

// Тип Вселенной (0-типов), где пути — это биекции/эквивалентности
type Universe = Any

given Groupoid[Universe] with
  type Id[X <: Universe, Y <: Universe] = Equiv[X, Y]
  def refl[X <: Universe]: Equiv[X, X] = Equiv(identity, identity)
  def sym[X <: Universe, Y <: Universe](e: Equiv[X, Y]): Equiv[Y, X] = Equiv(e.from, e.to)
  def trans[X <: Universe, Y <: Universe, Z <: Universe](e1: Equiv[X, Y], e2: Equiv[Y, Z]): Equiv[X, Z] =
    Equiv(e1.to.andThen(e2.to), e2.from.andThen(e1.from))

@main def proveFundamentalGroup(): Unit =
  val G_S1 = summon[Groupoid[S1]]
  val G_U  = summon[Groupoid[Universe]]

  // Определяем петлю во Вселенной: автоморфизм (x => x + 1) на типе Int
  val shiftEquiv: Equiv[Int, Int] = Equiv(_ + 1, _ - 1)

  // Применяем универсальный рекурсор S1.rec к Вселенной типов!
  // Получаем отображение code: S1 => Universe и его действие на путях transport
  val (code, transport) = S1.rec[Universe](Int, shiftEquiv)

  // Сконструируем произвольный сложный путь на окружности:
  // p = loop ∙ loop ∙ loop⁻¹ ∙ loop
  val p: S1.Loop = G_S1.trans(
    G_S1.trans(G_S1.trans(S1.loop, S1.loop), G_S1.sym(S1.loop)),
    S1.loop
  )

  // Вычисляем число оборотов (encode), транспортируя 0 вдоль пути p:
  val equivalenceAtP: Equiv[Int, Int] = transport(p).asInstanceOf[Equiv[Int, Int]]
  val windingNumber: Int = equivalenceAtP.to(0)

  println(s"Вычисленное число оборотов для p: $windingNumber") // Выведет: 2
